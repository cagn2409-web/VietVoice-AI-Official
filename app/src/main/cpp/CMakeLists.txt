cmake_minimum_required(VERSION 3.22.1)
project(vietvoice_hy LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp llama)
if(ANDROID)
  add_library(vietvoice_hy SHARED hy_jni.cpp)
  target_link_libraries(vietvoice_hy PRIVATE llama android log)
  target_link_options(vietvoice_hy PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(hy_smoke hy_smoke.cpp)
  target_link_libraries(hy_smoke PRIVATE llama)
endif()
